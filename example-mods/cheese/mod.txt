// ---------------------------------------------------------------- blocks
[block wheel]
name = Cheese Wheel
top = wheel_top
bottom = wheel_top
side = wheel_side.png
hardness = 0.8
sound = wood
// Landing on it bounces you back up (sneak to land softly).
bounce = 0.85

[block butter]
name = Block of Butter
texture = butter
hardness = 0.4
sound = sand
// Walk on it to go fast.
speed = 1.8

[block cheese_ore]
name = Cheese Ore
texture = cheese_ore
hardness = 2.5
tool = pickaxe
tier = 1
drops = slice
on_break = message The ore smells... aged.

[block cheese_flower]
name = Cheese Flower
model = cross
texture = cheese_flower
hardness = 0
drops = slice

[block stinky_cheese]
name = Stinky Cheese (do not break)
texture = wheel_top
hardness = 1
on_break = message That was a mistake.; explode 2.5

// ---------------------------------------------------------------- items
[item slice]
name = Cheese Slice
texture = slice
food = 3

[item cheese_pickaxe]
name = Cheese Pickaxe (Surprisingly Hard)
texture = iron_pickaxe
stack = 1
pickaxe = 3
damage = 5

[item launcher]
name = Wheel of Launching
texture = slice
stack = 1
consume = false
on_use = launch 18; message Wheeeeee!

[item summoning_brie]
name = Summoning Brie
texture = slice
// Spawns the cheese mouse defined below (any mob name works here).
on_use = spawn cheese_mouse; message A mouse smelled the brie.

// ---------------------------------------------------------------- mobs
// A new mob type: a passive wanderer wearing our slice texture. It never
// spawns on its own; the Summoning Brie above spawns one, and a script could
// too with spawn_mob("cheese_mouse", x, y, z). See MODDING.md for every key.
[mob cheese_mouse]
name = Cheese Mouse
texture = slice
template = quadruped
size = 0.4
health = 6
speed = 1.4
drops = slice 1

// ---------------------------------------------------------------- recipes
// Inputs: up to 6 "item count" pairs. Base-game items use their plain names.
[recipe]
inputs = slice 4
output = wheel 1

[recipe]
inputs = wheel 1
output = slice 4

[recipe]
inputs = slice 2, sand 1
output = butter 2

[recipe]
inputs = wheel 3, stick 2
output = cheese_pickaxe 1

[recipe]
inputs = wheel 1, gunpowder 1
output = launcher 1

[recipe]
inputs = slice 1, porkchop 1
output = summoning_brie 2

[recipe]
inputs = wheel 1, dirt 1
output = stinky_cheese 1

// ---------------------------------------------------------------- world generation
[ore cheese_vein]
block = cheese_ore
replace = stone
min_y = 5
max_y = 50
chance = 0.006

[plant cheese_flowers]
block = cheese_flower
on = grass
chance = 0.01
