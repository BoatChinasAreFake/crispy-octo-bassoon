// ---------------------------------------------------------------- mobs
// A new mob type: a passive wanderer wearing our slice texture. It never
// spawns on its own; the Summoning Brie above spawns one, and a script could
// too with spawn_mob("cheese_mouse", x, y, z). See MODDING.md for every key.
[mob cheese_mouse]
name = Cheese Mouse
texture = butter
template = quadruped
size = 0.4
health = 6
speed = 1.4
drops = slice 1

// A hostile mob that actually attacks: mark it hostile and give it an
// attack_damage, and it will pursue and bite nearby players, host-side, like a
// base-game monster. Leave attack_damage off (0) and a hostile mob just counts
// toward the monster cap, as before. See the [mob] key table in MODDING.md.
[mob angry_cheese]
name = Angry Cheese
texture = wheel_top
template = biped
width = 0.7
height = 1.9
health = 24
speed = 1.1
hostile = true
// bites for 5 (clamped to 0-50); reach is how close it must be (0.5-4);
// aggro_range is how far it notices you (1-48); cooldown is seconds per bite (min 0.25).
attack_damage = 5
attack_reach = 1.6
aggro_range = 20
attack_cooldown = 1.2
drops = slice 2

// A hostile mob that fights at range. Its three-shot fan reuses the slice
// texture as a bounded billboard and gives a short Leaping effect on a hit.
// It also keeps a small attack_damage, so it bites up close (melee wins inside
// attack_reach); melee and the whole ranged volley share one cooldown.
[mob cheese_slinger]
name = Cheese Slinger
texture = butter
template = biped
width = 0.7
height = 1.8
health = 20
speed = 1.0
hostile = true
ranged_damage = 4
ranged_range = 18
projectile_speed = 26
projectile_model = billboard
projectile_texture = slice
projectile_scale = 0.75
projectile_effect = leaping
projectile_effect_duration = 4
projectile_effect_amplifier = 0
projectile_count = 3
projectile_spread = 18
ranged_cooldown = 2.0
// also bites a little up close, to show the melee + ranged combo.
attack_damage = 3
attack_reach = 1.5
aggro_range = 24
drops = slice 2

// A flier you can tame and breed. It cruises five blocks up, lands for a
// rest now and then (perches), and follows anyone holding a slice. Feed it
// slices: half the time it's yours (it sits; right-click to call it along),
// and two tamed bats fed slices have a baby bat, born tame.
[mob cheese_bat]
name = Cheese Bat
texture = butter
template = bird
size = 0.5
health = 8
flying = true
fly_height = 5
fly_speed = 4
perches = true
tame_item = slice
tame_chance = 0.5
breed_item = slice

// Its shots curve gently toward you (homing, degrees per second) and burst
// when they land, hurting everyone within 2.5 blocks (blocks are never harmed).
[mob cheese_seeker]
name = Stinky Seeker
texture = wheel_top
template = blob
size = 0.8
health = 16
hostile = true
ranged_damage = 3
ranged_range = 20
projectile_model = billboard
projectile_texture = slice
projectile_homing = 60
projectile_blast = 2.5
ranged_cooldown = 3
aggro_range = 24
drops = slice 1

// A body built from its own boxes instead of a template. Each part is
// name x y z width height depth (blocks, from its feet; -z is the front),
// then options: pivot=x,y,z (where it turns; default its top middle),
// anim=still|walk|sway|wing|flap|tilt|bob|spin, amount=N (how much; minus
// swings the other way; degrees for tilt), texture=, face= (front), top=.
// Up to 16 parts. Its hitbox still comes from size/width/height.
[mob cheese_crab]
name = Cheddar Crab
texture = butter
size = 0.6
width = 1.0
health = 10
speed = 0.8
breed_item = slice
part = shell -0.4 0.25 -0.3 0.8 0.3 0.6 anim=bob amount=0.5 top=wheel_top
part = eyes -0.25 0.55 -0.3 0.5 0.12 0.1 face=wheel_top
part = claw_l -0.75 0.3 -0.55 0.3 0.2 0.35 pivot=-0.45,0.4,-0.3 anim=sway amount=1.2
part = claw_r 0.45 0.3 -0.55 0.3 0.2 0.35 pivot=0.45,0.4,-0.3 anim=sway amount=-1.2
part = leg_l1 -0.6 0 -0.15 0.25 0.3 0.08 pivot=-0.4,0.3,-0.1 anim=walk amount=1
part = leg_l2 -0.6 0 0.1 0.25 0.3 0.08 pivot=-0.4,0.3,0.14 anim=walk amount=-1
part = leg_r1 0.35 0 -0.15 0.25 0.3 0.08 pivot=0.4,0.3,-0.1 anim=walk amount=-1
part = leg_r2 0.35 0 0.1 0.25 0.3 0.08 pivot=0.4,0.3,0.14 anim=walk amount=1

// A trader. Right-click it to open the same trading screen as a Hmmer.
// trade = what you give [count] [+ a second thing [count]] -> what you get [count]
// ("for" works instead of "->"). Up to 8 trades, each 6 times a day.
[mob cheesemonger]
name = Cheesemonger
texture = butter
template = biped
width = 0.7
height = 1.8
health = 20
trade = slice 8 -> gold 1
trade = gold 2 -> wheel 1
trade = gold 3 + slice 4 -> cheese_pickaxe

// A boss: a health bar across the top of the screen, never tamed or
// despawned, and everyone hears when it's beaten. Below half health it
// enrages (faster, quicker to strike) and calls Angry Cheese for help.
[mob big_cheese]
name = The Big Cheese
texture = wheel_top
template = biped
width = 1.4
height = 2.8
health = 200
speed = 0.9
hostile = true
attack_damage = 7
attack_reach = 2.2
aggro_range = 28
attack_cooldown = 1.5
boss = true
enrage_at = 0.5
enrage_speed = 1.6
enrage_cooldown = 0.6
summon = angry_cheese 2
summon_every = 20
xp = 200
drops = slice 32
